\chapter*{DANH MỤC TỪ VIẾT TẮT}
\addcontentsline{toc}{chapter}{DANH MỤC TỪ VIẾT TẮT}

\begin{longtable}{@{}p{0.8in}p{4.7in}@{}}
  ACT    & Action Chunking with Transformers -- phân đoạn hành động với Transformer \\[4pt]
  CVAE   & Conditional Variational Autoencoder -- bộ tự mã hóa biến phân có điều kiện \\[4pt]
  DART   & Disturbances for Augmenting Robot Trajectories -- bơm nhiễu vào trình diễn \\[4pt]
  GRU    & Gated Recurrent Unit -- đơn vị hồi quy có cổng \\[4pt]
  HABIT  & Human-Aware Behavior and Interaction Training dataset -- bộ dữ liệu tương tác người -- robot \\[4pt]
  HMM    & Hidden Markov Model -- mô hình Markov ẩn \\[4pt]
  HRI    & Human--Robot Interaction -- tương tác người -- robot \\[4pt]
  JEPA   & Joint-Embedding Predictive Architecture -- kiến trúc dự đoán nhúng chung \\[4pt]
  PIF    & Predictive Intent Filter -- bộ lọc ý định dự đoán (mô hình đề xuất) \\[4pt]
  RANSAC & Random Sample Consensus -- đồng thuận mẫu ngẫu nhiên \\[4pt]
  RQ     & Research Question -- câu hỏi nghiên cứu \\[4pt]
  SIGReg & Sketched Isotropic Gaussian Regularization -- chuẩn hóa Gauss đẳng hướng trên phép chiếu \\[4pt]
  VICReg & Variance-Invariance-Covariance Regularization \\
\end{longtable}
