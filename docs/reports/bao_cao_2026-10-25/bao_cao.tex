\muc{1. Mục tiêu}
\begin{itemize}
  \item \textbf{Mục tiêu 1:} dùng dự báo chuyển động người, robot hỗ trợ \textbf{sớm hơn} khi dự báo đúng.
  \item \textbf{Mục tiêu 2:} robot \textbf{biết sửa} và vẫn hoàn thành khi dự báo sai hoặc người đổi ý.
  \item \textbf{Mục tiêu 3:} xây dựng \textbf{bộ dataset mô phỏng HRI} cho các trường hợp tương tác trên.
\end{itemize}
Các trường hợp test:
\begin{itemize}
  \item \textbf{C1} dự báo đúng và rõ;
  \item \textbf{C2} ý định mơ hồ;
  \item \textbf{C3} dự báo sai (chuyển động trông giống yêu cầu nhưng không phải);
  \item \textbf{C4} người đổi ý sau khi robot đã bắt đầu.
\end{itemize}

\muc{2. Kết quả hiện tại}
Đánh giá trên 120 tập mới (30 tập mỗi trường hợp, seed không trùng dữ liệu train).
CSR: tỷ lệ thành công; CT: thời gian hoàn thành (tập thành công);
TSync: độ lệch giữa lúc người sẵn sàng nhận và lúc robot tới chỗ trao (thời gian người phải chờ);
HCS: không va chạm và không vi phạm khoảng cách an toàn.

\begin{itemize}
  \item D hoàn thành nhanh hơn A ở cả 4 trường hợp, trung bình khoảng 3.4\,s (21\,\%); thời gian người phải chờ

  \item Khi dự báo sai (C3) và khi người đổi ý (C4), D vẫn hoàn thành đúng 100\,\% và nhanh hơn A.
  \item D hụt một số tập ở C1, C2 (93\,\%, 87\,\%), trong khi A đạt 100\,\% nhưng chậm hơn.
  \item Không có va chạm hay vi phạm khoảng cách an toàn ở cả hai mô hình.
\end{itemize}

\textbf{Giới hạn:} kết quả trong mô phỏng khả quan hơn thực tế: tập test gần với tập train, tư thế người lấy chính
xác từ mô phỏng, chuyển động người dùng chung thư viện, gắp có trợ giúp, hình ảnh không thay đổi.

\muc{3. Hướng triển khai tiếp}
\begin{itemize}
  \item Thu thêm dữ liệu để test ngoài phân phối: chuyển động người không có trong train, ánh sáng / màu khác,
  tư thế người có nhiễu như bộ ước lượng thật.
  \item Mở rộng sang các task cộng tác khác và train trên nhiều task.
  \item Train mô hình đối chứng (demo sớm, không dự báo) để tách phần lợi do dự báo ý định.
  \item Lặp lại với nhiều lần train để có khoảng tin cậy; sau đó thử trên robot thật.
\end{itemize}
