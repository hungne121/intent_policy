% Cỡ chữ sơ đồ cố định (không theo cỡ chữ thân bài 12pt): \figf = chữ khung 8pt, \figs = chữ phụ 7pt.
\providecommand{\figf}{\fontsize{8}{9.6}\selectfont}
\providecommand{\figs}{\fontsize{7}{8.4}\selectfont}

\definecolor{cPart}{HTML}{E65100}\definecolor{cPartBg}{HTML}{FFF3E0}

  part/.style={box, draw=cPart, fill=cPartBg},
  todo/.style={box, draw=cTodo, fill=cTodoBg, dashed},
